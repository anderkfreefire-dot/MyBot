@bot.callback_query_handler(func=lambda call: call.data == "create_user")
def ask_devices_and_duration(call):
  msg = bot.send_message(call.message.chat.id, "أدخل تفاصيل الاشتراك بهذا الشكل:\n(عدد الأجهزة وعدد أيام المدة)\nمثال: `1 30`")
  bot.register_next_step_handler(msg, process_user_creation)

def process_user_creation(message):
  try:
    args = message.text.split()
    devices = int(args[0])
    days = int(args[1])
    username, password = generate_credentials()
    DATABASE[username] = {"password": password, "devices": devices, "days": days, "active": True}
    response = f"✅ **تم إنشاء الحساب بنجاح!**\n\n👤 **اليوزر:** `{username}`\n🔑 **الباسورد:** `{password}`\n📱 **عدد الأجهزة:** `{devices}`\n⏳ **المدة:** `{days} يوم`"
    bot.send_message(message.chat.id, response, parse_mode="Markdown")
  except Exception as e:
    bot.send_message(message.chat.id, "❌ خطأ في التنسيق. اكتب هكذا (مثال: `1 30`).")

@bot.callback_query_handler(func=lambda call: call.data == "manage_users")
def list_users_for_management(call):
  if not DATABASE:
    bot.answer_callback_query(call.id, "لا توجد يوزرات حتى الآن.")
    return
  markup = types.InlineKeyboardMarkup(row_width=1)
  for user, info in DATABASE.items():
    status = "🟢 مفعل" if info["active"] else "🔴 متوقف"
    markup.add(types.InlineKeyboardButton(f"{user} ({status})", callback_data=f"toggle_{user}"))
  bot.edit_message_text("اختر اليوزر لتغيير حالته:", chat_id=call.message.chat.id, message_id=call.message.message_id, reply_markup=markup)

@bot.callback_query_handler(func=lambda call: call.data.startswith("toggle_"))
def toggle_user_status(call):
  username = call.data.replace("toggle_", "")
  if username in DATABASE:
    DATABASE[username]["active"] = not DATABASE[username]["active"]
    bot.answer_callback_query(call.id, "تم تغيير الحالة بنجاح")
    list_users_for_management(call)
